cmake_minimum_required(VERSION 3.20)
project(myMcpTools VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(mymcp STATIC src/mcp.cpp src/boost_json.cpp)
target_include_directories(mymcp PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(mymcp PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
target_link_libraries(mymcp PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(mymcp PUBLIC ws2_32 wsock32)
endif()
add_executable(mymcp_example_server examples/server.cpp)
target_link_libraries(mymcp_example_server PRIVATE mymcp)
add_executable(mymcp_example_client examples/client.cpp)
target_link_libraries(mymcp_example_client PRIVATE mymcp)
enable_testing()
add_executable(mymcp_tests tests/test_protocol.cpp)
target_link_libraries(mymcp_tests PRIVATE mymcp)
add_test(NAME protocol COMMAND mymcp_tests)
install(TARGETS mymcp EXPORT mymcpTargets ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT mymcpTargets NAMESPACE mymcp:: DESTINATION lib/cmake/mymcp)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME http_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/http_smoke.py" "$<TARGET_FILE:mymcp_example_server>")
endif()
