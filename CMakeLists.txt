cmake_minimum_required(VERSION 3.20)
project(myMcpTools VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(mymcp STATIC src/mcp.cpp src/boost_json.cpp src/schema.cpp src/oauth.cpp)
target_include_directories(mymcp PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(mymcp PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
target_link_libraries(mymcp PUBLIC Threads::Threads OpenSSL::SSL OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(mymcp PUBLIC ws2_32 wsock32)
endif()
add_executable(mymcp_example_server examples/server.cpp)
target_link_libraries(mymcp_example_server PRIVATE mymcp)
add_executable(mymcp_example_client examples/client.cpp)
target_link_libraries(mymcp_example_client PRIVATE mymcp)
add_executable(mymcp_listen_client tests/listen_client.cpp)
target_link_libraries(mymcp_listen_client PRIVATE mymcp)
enable_testing()
add_executable(mymcp_tests tests/test_protocol.cpp)
target_link_libraries(mymcp_tests PRIVATE mymcp)
add_test(NAME protocol COMMAND mymcp_tests)
add_executable(mymcp_extended_tests tests/test_extended.cpp)
target_link_libraries(mymcp_extended_tests PRIVATE mymcp)
add_test(NAME extended COMMAND mymcp_extended_tests)
add_executable(mymcp_tasks_schema_tests tests/test_tasks_schema.cpp)
target_link_libraries(mymcp_tasks_schema_tests PRIVATE mymcp)
add_test(NAME tasks_schema COMMAND mymcp_tasks_schema_tests)
add_executable(mymcp_oauth_tests tests/test_oauth.cpp)
target_link_libraries(mymcp_oauth_tests PRIVATE mymcp)
add_test(NAME oauth COMMAND mymcp_oauth_tests)
install(TARGETS mymcp EXPORT mymcpTargets ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT mymcpTargets NAMESPACE mymcp:: DESTINATION lib/cmake/mymcp)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME http_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/http_smoke.py" "$<TARGET_FILE:mymcp_example_server>")
  add_test(NAME stdio_subscription COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/stdio_subscription.py" "$<TARGET_FILE:mymcp_example_server>")
  add_test(NAME auth_http COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/auth_http.py" "$<TARGET_FILE:mymcp_example_server>")
  add_test(NAME progress_cancel COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/progress_cancel.py" "$<TARGET_FILE:mymcp_example_server>")
  add_test(NAME sse_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/sse_smoke.py" "$<TARGET_FILE:mymcp_example_server>" "$<TARGET_FILE:mymcp_listen_client>")
endif()
